INSERT INTO users (name, access_code, "longName")
SELECT
  'user' || LPAD(gs::text, 4, '0') AS name,
  SUBSTRING(MD5(RANDOM()::text) FROM 1 FOR 8) AS access_code,
  'Usuario ' || gs AS "longName"
FROM generate_series(1, 1000) AS gs;


--validar si hay codigos de acceso repetido
SELECT access_code, COUNT(*) AS veces
FROM users
GROUP BY access_code
HAVING COUNT(*) > 1;
